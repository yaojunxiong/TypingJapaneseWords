### 1.4 存储实现

**文件**：`src/components/lesson-conversation-client.tsx`

```typescript
const FAMILIARITY_KEY = 'minna.conversation.familiarity.v1'

function readFamiliarity(): Record<string, FamiliarityRecord> {
  if (typeof window === 'undefined') return {}
  try {
    return JSON.parse(localStorage.getItem(FAMILIARITY_KEY) || '{}')
  } catch { return {} }
}

function writeFamiliarity(data: Record<string, FamiliarityRecord>) {
  try {
    localStorage.setItem(FAMILIARITY_KEY, JSON.stringify(data))
    localStorage.setItem('minna.cloud.state.dirty_at.v1', String(Date.now()))
  } catch {}
}
```

**验收标准**：刷新页面后背诵进度不丢失。

## 阶段 2：我会了 / 不熟逻辑

### 2.1 不熟句子循环复习

在同一课 session 内，所有标记 `unfamiliar` 的句子会在全部句子完成后重新出现，每轮筛选用户最多标记 3 次 `unfamiliar` 后强制暂停该句。

```typescript
function getSessionQueue(items: ConversationItem[]): ConversationItem[] {
  const fam = readFamiliarity()
  const unfamiliar = items.filter(i => fam[i.id]?.status === 'unfamiliar' && (fam[i.id]?.count || 0) < 3)
  if (!unfamiliar.length) return items
  return [...items, ...unfamiliar]
}
```

**验收标准**：标记「不熟」的句子在同一课 session 末尾重新出现。
